memory_reader.cpp 
#include <Windows.h>
#include <iostream>

template<typename T>
T ReadMemory(HANDLE proc, uintptr_t addr) {
  T val{};
  ReadProcessMemory(proc,
    (LPCVOID)addr,
    &val, sizeof(T), nullptr);
  return val;
}

float hp = ReadMemory<float>(
  hProc, 0x1A2B3C
);
